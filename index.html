💻 Exemples de code

const { DocuTechClient } = require('docutech-sdk');
const client = new DocuTechClient({ apiKey: 'docu_sk_live_xxxx' });
const docs = await client.documents.list();
from docutech import Client
client = Client(api_key="docu_sk_live_xxxx")
documents = client.documents.list()
curl -X GET https://api.docutech.io/v2/documents \
