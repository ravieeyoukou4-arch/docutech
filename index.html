Exemples de code

// JavaScript - Authentification et requête
const { DocuTechClient } = require('docutech-sdk');

async function main() {
  const client = new DocuTechClient({
    apiKey: process.env.DOCUTECH_API_KEY
  });
  
  const docs = await client.documents.list();
  console.log(`📄 ${docs.length} documents disponibles`);
}

main().catch(console.error);
# Python - SDK DocuTech
from docutech import Client
import os

client = Client(api_key=os.getenv("DOCUTECH_API_KEY"))

# Récupérer les documents
documents = client.documents.list()
for doc in documents:
    print(f"📄 {doc.title} - {doc.version}")

# Recherche
results = client.search("hardware configuration")
print(f"🔍 {len(results)} résultats")
# cURL - Appel direct à l'API
curl -X GET https://api.docutech.io/v2/documents \
  -H "Authorization: Bearer docu_sk_live_xxxxxxxxxxxx" \
  -H "Content-Type: application/json"
